HomeLink v2.6 Complete Feature Guide & How-To

Offline Icon Caching & Sync

Browse online icon lookups or favicons normally. When you save a tile, the server automatically downloads and caches the asset to persistent storage (`/data/icons/`), making it available offline and synced instantly across all your devices.

Heimdall JSON Migration

Switching from Heimdall? Go to Settings and click Import Heimdall. HomeLink parses your old export JSON, mapping titles, URLs, colors, and API keys instantly!

Direct URL Bookmarking

Open the Profiles modal and click Direct Link next to any profile to copy its unique bookmark URL (`?profile=id`) to bypass the selection screen.

Password Protected Management

Unlocking edit mode, updating passwords, editing profiles, and creating/deleting users automatically prompt for password authentication if enabled (locking back down is instant).

Docker Deployment & Complete File Setup Guide

To synchronize your configuration and cached icons across all devices, HomeLink uses a Node.js backend that persists state to a mounted Docker volume (`/app/data/`). Copy and paste the contents below into your project files:

1. `server.js` (Backend API, Static Server & Icon Caching Proxy)
const express = require('express');
const fs = require('fs');
const path = require('path');
const https = require('https');
const http = require('http');

const app = express();
const PORT = process.env.PORT || 8580;
const DATA_DIR = path.join(__dirname, 'data');
const CONFIG_FILE = path.join(DATA_DIR, 'config.json');
const ICONS_DIR = path.join(DATA_DIR, 'icons');

if (!fs.existsSync(DATA_DIR)) {
    fs.mkdirSync(DATA_DIR, { recursive: true });
}
if (!fs.existsSync(ICONS_DIR)) {
    fs.mkdirSync(ICONS_DIR, { recursive: true });
}

app.use(express.json({ limit: '10mb' }));
app.use(express.static(__dirname));
app.use('/data', express.static(DATA_DIR));

function downloadAndCacheIcon(url) {
    return new Promise((resolve) => {
        if (!url || (!url.startsWith('http://') && !url.startsWith('https://'))) {
            return resolve(url);
        }
        const filename = Buffer.from(url).toString('base64').replace(/[/+=]/g, '_') + '.png';
        const localPath = path.join(ICONS_DIR, filename);
        const relativePath = `/data/icons/${filename}`;

        if (fs.existsSync(localPath)) {
            return resolve(relativePath);
        }

        const client = url.startsWith('https') ? https : http;
        const request = client.get(url, (response) => {
            if (response.statusCode >= 300 && response.statusCode < 400 && response.headers.location) {
                return downloadAndCacheIcon(response.headers.location).then(resolve);
            }
            if (response.statusCode !== 200) {
                return resolve(url);
            }
            const fileStream = fs.createWriteStream(localPath);
            response.pipe(fileStream);
            fileStream.on('finish', () => {
                fileStream.close();
                resolve(relativePath);
            });
            fileStream.on('error', () => resolve(url));
        });
        request.on('error', () => resolve(url));
        request.setTimeout(5000, () => { request.destroy(); resolve(url); });
    });
}

async function cacheAllExistingIconsOnStartup() {
    if (!fs.existsSync(CONFIG_FILE)) return;
    try {
        const raw = fs.readFileSync(CONFIG_FILE, 'utf8');
        const config = JSON.parse(raw);
        let updated = false;
        if (config && Array.isArray(config.profiles)) {
            for (const profile of config.profiles) {
                if (Array.isArray(profile.apps)) {
                    for (const appItem of profile.apps) {
                        if (appItem.icon && (appItem.icon.startsWith('http://') || appItem.icon.startsWith('https://'))) {
                            const cachedPath = await downloadAndCacheIcon(appItem.icon);
                            if (cachedPath !== appItem.icon) {
                                appItem.icon = cachedPath;
                                updated = true;
                            }
                        }
                    }
                }
            }
        }
        if (updated) {
            fs.writeFileSync(CONFIG_FILE, JSON.stringify(config, null, 2), 'utf8');
            console.log('[Icon Cache] Startup migration complete: All existing external icons cached.');
        }
    } catch (err) {
        console.error('[Icon Cache] Startup scan failed:', err.message);
    }
}

app.get('/api/config', (req, res) => {
    if (fs.existsSync(CONFIG_FILE)) {
        try {
            const data = fs.readFileSync(CONFIG_FILE, 'utf8');
            res.json(JSON.parse(data));
        } catch (err) {
            res.status(500).json({ error: 'Failed to parse config file' });
        }
    } else {
        res.json({ categories: [], tiles: [] });
    }
});

app.post('/api/config', (req, res) => {
    try {
        fs.writeFileSync(CONFIG_FILE, JSON.stringify(req.body, null, 2), 'utf8');
        res.json({ success: true });
    } catch (err) {
        res.status(500).json({ error: 'Failed to save config file' });
    }
});

app.post('/api/cache-icon', async (req, res) => {
    const { url } = req.body;
    if (!url || (!url.startsWith('http://') && !url.startsWith('https://'))) {
        return res.status(400).json({ error: 'Invalid URL' });
    }
    const cachedUrl = await downloadAndCacheIcon(url);
    res.json({ url: cachedUrl });
});

cacheAllExistingIconsOnStartup().then(() => {
    app.listen(PORT, () => {
        console.log(`HomeLink running on port ${PORT}`);
    });
});
